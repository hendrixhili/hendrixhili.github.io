\section{Research Experience}
\entry{K-Band Reconfigurable Intelligent Surface (RIS) for Millimeter-Wave Systems}{Dec 2024 -- Dec 2025}
  {\textit{Prof.\ Shaoqiu Xiao, Sun Yat-sen University}}{}
\begin{itemize}
  \item Designed a multilayer K-band RIS unit cell (20--26\,GHz) featuring two rotationally symmetric hollow I-shaped resonant layers on RO3003 substrate, optimized via HFSS with Floquet-port simulation; unit size 4$\times$4\,mm ($\approx$0.30$\lambda$ at 22\,GHz)
  \item Achieved $\sim$360$^\circ$ phase tuning range with $\sim$120$^\circ$ step per coding state, reflection loss $<$1.6\,dB for four primary states, and excellent phase linearity ($R^2 = 0.9991$, RMSE $= 0.515^\circ$) within the 22--23\,GHz operating band
  \item First-author paper published at EuCAP 2026, Dublin (see Publications)
\end{itemize}
